语法速览

Vora 的语法受到 Lua 和 Wren 的启发，追求简洁与可读性。

变量

let name = "Vora"
let version = 0.1
let active = TRUE

字符串插值

let name = "World"
print("Hello ${name}!")
// → Hello World!

函数

func add(a, b) {
    return a + b
}

let result = add(3, 7)
print(result)  // → 10

条件判断

if (score >= 90) {
    print("优秀")
} else if (score >= 60) {
    print("及格")
} else {
    print("不及格")
}

循环

// while 循环
let i = 0
while (i < 5) {
    print(i)
    i = i + 1
}

// for-in 循环
for item in [1, 2, 3] {
    print(item)
}

对象

Obj Student(name, age) {
    this.name = name
    this.age = age

    func greet() {
        print("Hi, I'm ${this.name}")
    }
}

let s = Student("Alice", 20)
s.greet()  // → Hi, I'm Alice
